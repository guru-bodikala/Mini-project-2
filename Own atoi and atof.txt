double my_atof(char *s)
{
	int i=0;
	double add=0,p=0.1,sign=1;
	if(s[0]=='-')
	{
		sign=-1;
		i=1;
	}
	for( ;s[i];i++)
	{
		if(s[i]=='.')
		{
			i++;
			break;
		}
		else if(s[i]>='0'&& s[i]<='9')
		{
			add=add*10+(s[i]-'0');
		}
	}
	for( ;s[i];i++)
	{
		if(s[i]>='0' && s[i]=='9')
		{
			add=add+(s[i]-'0')*p;
			p/=10;
		}
		else
			break;
	}
	return sign*add;
}

int my_atoi(char *s)
{
	int i=0,add=0,sign=1;
	if(*s=='-')
	{
		sign=-1;
		i=1;
	}
	for( ;s[i];i++)
	{
		if(s[i]>='0' && s[i]<='9')
			add=add*10+(s[i]-'0');
		else
			break;
	}
	return sign*add;
}
